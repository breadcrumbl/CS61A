# What Would Python Do?

# Q1

>>> s = [7//3, 5, [4, 0, 1], 2]
>>> s[0]
2

>>> s[2]
[4, 0, 1]

>>> s[-1]
2 
# Explanation: indexing wraps round. -n -> len-n

>>> len(s)
4

>>> 4 in s
False

>>> 4 in s[2]
True

>>> s[2] + [3 + 2]
[4, 0, 1, 5]

>>> 5 in s[2]
False

>>> s[2] * 2
[4, 0, 1, 4, 0, 1]
# Explanation: same as s[2] * s[2] which concats the lists

>>> list(range(3, 6))
[3, 4, 5]

>>> range(3, 6)
range(3, 6) 

>>> r = range(3, 6)
>>> [r[0], r[2]]
[3, 5]

>>> range(4)[-1]
3
# Explanation: range(4) -> 0, 1, 2, 3; -1 index wraps around to last entry


# Q4

>>> [2 * x for x in range(4)]
[0, 2, 4, 6]

>>> [y for y in [6, 1, 6, 1] if y > 2]
[6, 6]

>>> [[1] + s for s in [[4], [5, 6]]]
[ [1, 4], [1, 5, 6] ]

>>> [z + 1 for z in range(10) if z % 3 == 0]
[1, 4, 7, 10]
# Explanation: multiples of 3: [0, 3, 6, 9], then add 1
